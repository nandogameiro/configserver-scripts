# NOTE: download.configserver.com no longer exists. Do NOT follow any older
# instructions that fetch cmq.tgz from that host over http - the domain is out
# of the original author's hands and anything served from it would be executed
# as root by install.sh. Install from a source tree you have obtained and
# verified yourself.

# IMPORTANT: install from this cmq/ source directory, NOT from the cmq.tgz in
# the top level of the repository. That tarball is kept as the unmodified
# upstream artifact so its integrity can still be verified, and it does NOT
# contain the security fixes described in SECURITY-NOTES.md.

# To install or upgrade cmq, from the root shell via SSH, run the installer
# from inside this directory:

cd /path/to/configserver-scripts/cmq
sh install.sh

# If you want to uninstall, simply:

cd /etc/cmq
sh uninstall.sh
